name: Contrôle quotidien du site Easy Booking

# Vérifie chaque jour que le site fonctionne correctement et dépose un
# rapport dans la messagerie de l'espace administrateur.
# Ce workflow NE MODIFIE JAMAIS le site — détection uniquement.
on:
  schedule:
    - cron: '30 5 * * *'   # 05:30 UTC = 07:30 Paris en été, après la publication du jour
  workflow_dispatch: {}

jobs:
  check-site:
    runs-on: ubuntu-latest
    steps:
      - name: Récupérer le dépôt
        uses: actions/checkout@v4

      - name: Installer Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'

      - name: Exécuter le contrôle du site
        env:
          SITE_URL: ${{ vars.SITE_URL }}
          FIREBASE_PROJECT_ID: ${{ vars.FIREBASE_PROJECT_ID }}
          FIREBASE_API_KEY: ${{ secrets.FIREBASE_API_KEY }}
          ANTHROPIC_API_KEY: ${{ secrets.ANTHROPIC_API_KEY }}
          GOOGLE_SERVICE_ACCOUNT_JSON: ${{ secrets.GOOGLE_SERVICE_ACCOUNT_JSON }}
          GSC_SITE_URL: ${{ vars.GSC_SITE_URL }}
        run: node scripts/check-site.js
